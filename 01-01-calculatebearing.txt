/* ── Bearing calculation ─────────────────────────────────── */
function calculateBearing(
  lat1: number, lon1: number,
  lat2: number, lon2: number,
): number {
  const R = Math.PI / 180;
  const dLon = (lon2 - lon1) * R;
  const y = Math.sin(dLon) * Math.cos(lat2 * R);
  const x = Math.cos(lat1 * R) * Math.sin(lat2 * R) -
            Math.sin(lat1 * R) * Math.cos(lat2 * R) * Math.cos(dLon);
  const bearing = (Math.atan2(y, x) * 180 / Math.PI + 360) % 360;
  return Math.round(bearing);
}

/* ── Distance calculation ────────────────────────────────── */
function calculateDistance(
  lat1: number, lon1: number,
  lat2: number, lon2: number,
): number {
  const R    = 6371;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a    =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(lat1 * Math.PI / 180) *
    Math.cos(lat2 * Math.PI / 180) *
    Math.sin(dLon / 2) ** 2;
  return Math.round(R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)));
}

/* ── Bearing to compass ──────────────────────────────────── */
function bearingToCompass(bearing: number): string {
  const dirs = ["N","NNE","NE","ENE","E","ESE","SE","SSE",
                "S","SSW","SW","WSW","W","WNW","NW","NNW"];
  return dirs[Math.round(bearing / 22.5) % 16];
}

/* ── 3D Globe Component ──────────────────────────────────── */
function Globe3D({
  locations, selected, userLat, userLon, onSelect,
}: {
  locations:  GateLocation[];
  selected:   GateLocation | null;
  userLat:    number | null;
  userLon:    number | null;
  onSelect:   (loc: GateLocation) => void;
}) {
  const canvasRef  = useRef<HTMLCanvasElement>(null);
  const frameRef   = useRef(0);
  const rotationRef = useRef(0);
  const [hoveredId, setHoveredId] = useState<string | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d")!;
    const W   = canvas.offsetWidth;
    const H   = canvas.offsetHeight;
    canvas.width  = W * window.devicePixelRatio;
    canvas.height = H * window.devicePixelRatio;
    ctx.scale(window.devicePixelRatio, window.devicePixelRatio);

    const cx = W / 2;
    const cy = H / 2;
    const r  = Math.min(W, H) * 0.38;

    const latLonToXY = (lat: number, lon: number, rotation: number) => {
      const lonRad = (lon + rotation) * Math.PI / 180;
      const latRad = lat * Math.PI / 180;
      const x = cx + r * Math.cos(latRad) * Math.sin(lonRad);
      const y = cy - r * Math.sin(latRad);
      const z = Math.cos(latRad) * Math.cos(lonRad);
      return { x, y, z, visible: z > -0.2 };
    };

    const draw = () => {
      ctx.clearRect(0, 0, W, H);

      // Globe base
      const grad = ctx.createRadialGradient(cx - r * 0.3, cy - r * 0.3, 0, cx, cy, r);
      grad.addColorStop(0, "rgba(5,5,32,0.9)");
      grad.addColorStop(0.5, "rgba(0,0,15,0.95)");
      grad.addColorStop(1, "rgba(0,0,8,1)");
      ctx.beginPath();
      ctx.arc(cx, cy, r, 0, Math.PI * 2);
      ctx.fillStyle = grad;
      ctx.fill();

      // Globe border glow
      ctx.beginPath();
      ctx.arc(cx, cy, r, 0, Math.PI * 2);
      ctx.strokeStyle = "rgba(124,58,237,0.4)";
      ctx.lineWidth   = 2;
      ctx.shadowBlur  = 20;
      ctx.shadowColor = "#7c3aed";
      ctx.stroke();
      ctx.shadowBlur  = 0;

      // Latitude lines
      for (let lat = -60; lat <= 60; lat += 30) {
        ctx.beginPath();
        let first = true;
        for (let lon = 0; lon <= 360; lon += 5) {
          const { x, y, z } = latLonToXY(lat, lon, rotationRef.current);
          if (z > 0) {
            if (first) { ctx.moveTo(x, y); first = false; }
            else ctx.lineTo(x, y);
          } else {
            first = true;
          }
        }
        // 33rd parallel highlight
        ctx.strokeStyle = lat === 30
          ? "rgba(251,191,36,0.35)"
          : "rgba(124,58,237,0.08)";
        ctx.lineWidth   = lat === 30 ? 1.5 : 0.5;
        ctx.stroke();
      }

      // Longitude lines
      for (let lon = 0; lon < 360; lon += 30) {
        ctx.beginPath();
        let first = true;
        for (let lat = -80; lat <= 80; lat += 5) {
          const { x, y, z } = latLonToXY(lat, lon, rotationRef.current);
          if (z > 0) {
            if (first) { ctx.moveTo(x, y); first = false; }
            else ctx.lineTo(x, y);
          } else {
            first = true;
          }
        }
        ctx.strokeStyle = "rgba(124,58,237,0.06)";
        ctx.lineWidth   = 0.5;
        ctx.stroke();
      }

      // 33rd parallel label
      const p33 = latLonToXY(33, 90, rotationRef.current);
      if (p33.visible) {
        ctx.fillStyle = "rgba(251,191,36,0.5)";
        ctx.font      = "10px JetBrains Mono";
        ctx.fillText("33°N", p33.x + 5, p33.y);
      }

      // User location
      if (userLat !== null && userLon !== null) {
        const pos = latLonToXY(userLat, userLon, rotationRef.current);
        if (pos.visible) {
          ctx.beginPath();
          ctx.arc(pos.x, pos.y, 6, 0, Math.PI * 2);
          ctx.fillStyle   = "#22c55e";
          ctx.shadowBlur  = 15;
          ctx.shadowColor = "#22c55e";
          ctx.fill();
          ctx.shadowBlur  = 0;
          ctx.fillStyle   = "rgba(34,197,94,0.5)";
          ctx.font        = "9px JetBrains Mono";
          ctx.fillText("YOU", pos.x + 8, pos.y + 3);
        }
      }

      // Gate locations
      locations.forEach(loc => {
        const pos = latLonToXY(loc.lat, loc.lon, rotationRef.current);
        if (!pos.visible) return;

        const isSelected = selected?.id === loc.id;
        const isHovered  = hoveredId === loc.id;
        const size       = isSelected ? 10 : isHovered ? 8 : 6;

        // Pulse ring
        if (isSelected || loc.priority === "CRITICAL") {
          ctx.beginPath();
          ctx.arc(pos.x, pos.y, size * 2.5, 0, Math.PI * 2);
          ctx.strokeStyle = `${loc.color}40`;
          ctx.lineWidth   = 1;
          ctx.stroke();
        }

        // Pin
        ctx.beginPath();
        ctx.arc(pos.x, pos.y, size, 0, Math.PI * 2);
        ctx.fillStyle   = loc.color;
        ctx.shadowBlur  = isSelected ? 20 : 10;
        ctx.shadowColor = loc.color;
        ctx.fill();
        ctx.shadowBlur  = 0;

        // Label
        if (isSelected || isHovered) {
          ctx.fillStyle = "rgba(0,0,8,0.8)";
          const tw = ctx.measureText(loc.name).width;
          ctx.fillRect(pos.x + 12, pos.y - 10, tw + 8, 18);
          ctx.fillStyle = loc.color;
          ctx.font      = "bold 10px Space Grotesk";
          ctx.fillText(loc.name, pos.x + 16, pos.y + 3);
        }
      });

      // Connection line — user to selected
      if (selected && userLat !== null && userLon !== null) {
        const userPos = latLonToXY(userLat, userLon, rotationRef.current);
        const selPos  = latLonToXY(selected.lat, selected.lon, rotationRef.current);
        if (userPos.visible && selPos.visible) {
          ctx.beginPath();
          ctx.moveTo(userPos.x, userPos.y);
          ctx.lineTo(selPos.x, selPos.y);
          ctx.strokeStyle = `${selected.color}60`;
          ctx.lineWidth   = 1.5;
          ctx.setLineDash([4, 6]);
          ctx.stroke();
          ctx.setLineDash([]);
        }
      }

      rotationRef.current = (rotationRef.current + 0.15) % 360;
      frameRef.current    = requestAnimationFrame(draw);
    };

    draw();
    return () => cancelAnimationFrame(frameRef.current);
  }, [locations, selected, userLat, userLon, hoveredId]);

  // ── Click handler ─────────────────────────────────────────
  const handleClick = useCallback((e: React.MouseEvent<HTMLCanvasElement>) => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const rect  = canvas.getBoundingClientRect();
    const mx    = e.clientX - rect.left;
    const my    = e.clientY - rect.top;
    const W     = rect.width;
    const H     = rect.height;
    const cx    = W / 2;
    const cy    = H / 2;
    const r     = Math.min(W, H) * 0.38;

    // Find closest location to click
    let closest:  GateLocation | null = null;
    let closestD  = Infinity;

    locations.forEach(loc => {
      const lonRad = (loc.lon + rotationRef.current) * Math.PI / 180;
      const latRad = loc.lat * Math.PI / 180;
      const x      = cx + r * Math.cos(latRad) * Math.sin(lonRad);
      const y      = cy - r * Math.sin(latRad);
      const z      = Math.cos(latRad) * Math.cos(lonRad);
      if (z < -0.2) return;

      const d = Math.sqrt((mx - x) ** 2 + (my - y) ** 2);
      if (d < closestD && d < 25) {
        closestD = d;
        closest  = loc;
      }
    });

    if (closest) onSelect(closest);
  }, [locations, onSelect]);

  return (
    <canvas
      ref={canvasRef}
      className="w-full rounded-2xl cursor-crosshair"
      style={{ height: 480 }}
      onClick={handleClick}
      onMouseMove={e => {
        const canvas = canvasRef.current;
        if (!canvas) return;
        const rect = canvas.getBoundingClientRect();
        const mx   = e.clientX - rect.left;
        const my   = e.clientY - rect.top;
        const W    = rect.width;
        const H    = rect.height;
        const cx   = W / 2;
        const cy   = H / 2;
        const r    = Math.min(W, H) * 0.38;

        let found: string | null = null;
        locations.forEach(loc => {
          const lonRad = (loc.lon + rotationRef.current) * Math.PI / 180;
          const latRad = loc.lat * Math.PI / 180;
          const x      = cx + r * Math.cos(latRad) * Math.sin(lonRad);
          const y      = cy - r * Math.sin(latRad);
          const z      = Math.cos(latRad) * Math.cos(lonRad);
          if (z < -0.2) return;
          if (Math.sqrt((mx - x) ** 2 + (my - y) ** 2) < 15) found = loc.id;
        });
        setHoveredId(found);
      }}
      onMouseLeave={() => setHoveredId(null)}
    />
  );
}

      >
        {/* Cardinal directions */}
        {["N","E","S","W"].map((dir, i) => (

              left:      i === 3 ? 8 : i === 1 ? "auto" : "50%",

              transform: i === 0 || i === 2 ? "translateX(-50%)" : "translateY(-50%)",
            }}
          >
            {dir}
          </div>
        ))}

        {/* Rotating needle */}
        {bearing !== null && (
          <motion.div
            className="absolute"
            style={{ rotate: bearing }}
            animate={{ rotate: bearing }}
            transition={{ type: "spring", stiffness: 100 }}
          >
            {/* North needle — red */}
            <div style={{
              position:    "absolute",
              width:       3,
              height:      45,
              background:  "linear-gradient(180deg, #ef4444, transparent)",
              borderRadius: 2,
              left:        -1.5,

            }} />
            {/* South needle — white */}
            <div style={{
              position:    "absolute",
              width:       3,
              height:      35,
              background:  "linear-gradient(0deg, rgba(226,232,240,0.6), transparent)",
              borderRadius: 2,
              left:        -1.5,
              top:         0,
              transformOrigin: "top center",
            }} />
          </motion.div>
        )}

        {/* Center dot */}
        <div
          className="relative z-10 rounded-full"
          style={{ width: 8, height: 8, background: color }}

      </div>

      {/* Bearing readout */}
      {bearing !== null && (
        <div className="absolute -bottom-8 text-center">
          <div className="font-mono font-bold text-lg"
               style={{ color }}>
            {bearing}°

        </div>
      )}
    </div>

}

/* ── Tone player for location ────────────────────────────── */
function LocationTonePlayer({
  location,
}: {
  location: GateLocation;
}) {
  const ctxRef    = useRef<AudioContext | null>(null);
  const masterRef = useRef<GainNode | null>(null);
  const [playing, setPlaying] = useState<Set<number>>(new Set());
  const [volume,  setVolume]  = useState(0.6);
  const oscsRef   = useRef<Map<number, OscillatorNode[]>>(new Map());
  const gainsRef  = useRef<Map<number, GainNode>>(new Map());

  const init = useCallback(() => {
    if (ctxRef.current) return;
    const ctx    = new AudioContext({ sampleRate: 48000 });
    const master = ctx.createGain();
    master.gain.value = volume;
    master.connect(ctx.destination);
    ctxRef.current  = ctx;
    masterRef.current = master;
  }, [volume]);

  const playHz = useCallback((hz: number) => {
    init();
    const ctx    = ctxRef.current!;
    const master = masterRef.current!;

    if (playing.has(hz)) {
      // Stop
      const gain = gainsRef.current.get(hz);
      if (gain) {
        gain.gain.linearRampToValueAtTime(0, ctx.currentTime + 0.3);
        setTimeout(() => {
          oscsRef.current.get(hz)?.forEach(o => {
            try { o.stop(); o.disconnect(); } catch {}
          });
          gain.disconnect();
        }, 350);
      }
      setPlaying(prev => { const n = new Set(prev); n.delete(hz); return n; });
      return;
    }

    const gain = ctx.createGain();
    gain.gain.value = 0;
    gain.connect(master);

    // Binaural beat
    const merger = ctx.createChannelMerger(2);
    merger.connect(gain);
    const oscL = ctx.createOscillator();
    const oscR = ctx.createOscillator();
    const gL   = ctx.createGain();
    const gR   = ctx.createGain();
    oscL.type = oscR.type = "sine";
    oscL.frequency.value = hz;
    oscR.frequency.value = hz + (hz < 20 ? hz * 0.5 : 40);
    gL.gain.value = gR.gain.value = 0.5;
    oscL.connect(gL); gL.connect(merger, 0, 0);
    oscR.connect(gR); gR.connect(merger, 0, 1);
    oscL.start(); oscR.start();

    gain.gain.linearRampToValueAtTime(0.35, ctx.currentTime + 0.8);

    oscsRef.current.set(hz, [oscL, oscR]);
    gainsRef.current.set(hz, gain);
    setPlaying(prev => new Set(prev).add(hz));
  }, [playing, init]);

  const stopAll = useCallback(() => {
    const ctx = ctxRef.current;
    if (!ctx) return;
    playing.forEach(hz => {
      const gain = gainsRef.current.get(hz);
      if (gain) {
        gain.gain.linearRampToValueAtTime(0, ctx.currentTime + 0.3);
        setTimeout(() => {
          oscsRef.current.get(hz)?.forEach(o => {
            try { o.stop(); o.disconnect(); } catch {}
          });
          gain.disconnect();
        }, 350);
      }
    });
    setPlaying(new Set());
  }, [playing]);

  const playAll = useCallback(() => {
    stopAll();
    [location.brainLockHz, ...location.stackHz].forEach((hz, i) => {
      setTimeout(() => playHz(hz), i * 400);
    });
  }, [location, stopAll, playHz]);

  useEffect(() => () => { stopAll(); }, []);

  const allFreqs = [
    location.brainLockHz,
    ...location.stackHz.filter(hz => hz !== location.brainLockHz),
  ];

  return (
    <div className="rounded-2xl overflow-hidden"
         style={{
           background:     "rgba(0,0,8,0.8)",
           border:         `1px solid ${location.color}25`,
           backdropFilter: "blur(20px)",
         }}>
      {/* Header */}
      <div className="px-4 py-3 flex items-center justify-between"
           style={{ borderBottom: "1px solid rgba(255,255,255,0.05)" }}>
        <div className="flex items-center gap-2">
          <Volume2 size={14} style={{ color: location.color }} />
          <span className="text-xs font-mono font-bold"
                style={{ color: location.color }}>
            LOCATION FREQUENCIES
          </span>
          {playing.size > 0 && (
            <motion.div
              className="w-1.5 h-1.5 rounded-full"
              style={{ background: "#22c55e" }}
              animate={{ opacity: [1, 0.3, 1] }}
              transition={{ duration: 1, repeat: Infinity }}
            />
          )}
        </div>
        <div className="flex gap-2">
          <button
            onClick={playAll}
            className="text-[10px] font-mono px-2.5 py-1.5 rounded-lg
                       font-bold transition-all hover:scale-105"
            style={{
              background: `${location.color}20`,
              color:       location.color,
              border:     `1px solid ${location.color}40`,
            }}
          >
            ▶ Full Protocol
          </button>
          {playing.size > 0 && (
            <button
              onClick={stopAll}
              className="text-[10px] font-mono px-2.5 py-1.5 rounded-lg"
              style={{
                background: "rgba(239,68,68,0.12)",
                color:      "#ef4444",
                border:     "1px solid rgba(239,68,68,0.3)",
              }}
            >
              ■ Stop
            </button>
          )}
        </div>
      </div>

      {/* Volume */}
      <div className="px-4 py-2 flex items-center gap-3"
           style={{ borderBottom: "1px solid rgba(255,255,255,0.03)" }}>
        <Volume2 size={12} style={{ color: "rgba(226,232,240,0.3)" }} />
        <input
          type="range" min={0} max={1} step={0.01}
          value={volume}
          onChange={e => {
            const v = Number(e.target.value);
            setVolume(v);
            if (masterRef.current && ctxRef.current) {
              masterRef.current.gain.linearRampToValueAtTime(
                v, ctxRef.current.currentTime + 0.1
              );
            }
          }}
          className="flex-1"
          style={{ accentColor: location.color }}
        />
        <span className="text-[9px] font-mono w-6"
              style={{ color: "rgba(226,232,240,0.3)" }}>
          {Math.round(volume * 100)}%
        </span>
      </div>

      {/* Frequency buttons */}
      <div className="p-4 grid grid-cols-4 gap-2">
        {allFreqs.map((hz, i) => {
          const isOn = playing.has(hz);
          const isBrain = hz === location.brainLockHz;
          return (
            <motion.button
              key={hz}
              onClick={() => { init(); playHz(hz); }}
              whileHover={{ scale: 1.05 }}
              whileTap={{ scale: 0.95 }}
              className="relative rounded-xl py-2.5 px-1 text-center overflow-hidden"
              style={{
                background: isOn ? `${location.color}18` : "rgba(255,255,255,0.03)",
                border:     `1px solid ${isOn ? `${location.color}50` : "rgba(255,255,255,0.07)"}`,
                boxShadow:  isOn ? `0 0 10px ${location.color}20` : "none",
              }}
            >
              {isBrain && (
                <div className="absolute top-1 right-1 w-1.5 h-1.5
                                rounded-full bg-amber-400" />
              )}
              {isOn && (
                <motion.div
                  className="absolute inset-0 rounded-xl"
                  style={{ background: location.color }}
                  animate={{ opacity: [0.06, 0.14, 0.06] }}
                  transition={{ duration: 1.5, repeat: Infinity }}
                />
              )}
              <div className="relative z-10">
                <div className="font-mono font-bold text-xs"
                     style={{
                       color:      location.color,
                       textShadow: isOn ? `0 0 6px ${location.color}` : "none",
                     }}>
                  {hz}
                </div>
                <div className="text-[8px] text-white/30">Hz</div>
                {isOn && (
                  <div className="flex justify-center gap-0.5 mt-1">
                    {[1,2,3].map((_, j) => (
                      <motion.div
                        key={j}
                        className="w-0.5 rounded-t-sm"
                        style={{ background: location.color }}
                        animate={{ height: [2, 6, 2] }}
                        transition={{ duration: 0.5, delay: j * 0.1, repeat: Infinity }}
                      />
                    ))}
                  </div>
                )}
              </div>
            </motion.button>
          );
        })}
      </div>
    </div>
  );
}

                                 ? "rgba(239,68,68,0.15)"
                                 : selected.priority === "CLASSIFIED"
                                 ? "rgba(124,58,237,0.15)"
                                 : "rgba(245,158,11,0.15)",
                               color: selected.priority === "CRITICAL"
                                 ? "#ef4444"
                                 : selected.priority === "CLASSIFIED"
                                 ? "#a855f7"
                                 : "#f59e0b",
                               border: `1px solid ${selected.priority === "CRITICAL"
                                 ? "rgba(239,68,68,0.3)"
                                 : "rgba(245,158,11,0.3)"}`,
                             }}>
                          {selected.priority}
                        </div>
                        {selected.treaty && (
                          <div className="text-[9px] font-mono px-2 py-0.5 rounded"
                               style={{
                                 background: "rgba(239,68,68,0.1)",
                                 color:      "#ef4444",
                                 border:     "1px solid rgba(239,68,68,0.2)",
                               }}>
                            TREATY RESTRICTED
                          </div>
                        )}
                      </div>
                    </div>

                    {/* Coordinates */}
                    <div className="flex gap-3 mb-3">
                      <div className="flex-1 p-2 rounded-lg text-center"
                           style={{
                             background: "rgba(255,255,255,0.04)",
                             border:     "1px solid rgba(255,255,255,0.07)",
                           }}>
                        <div className="text-[9px] font-mono text-white/30">LATITUDE</div>
                        <div className="font-mono font-bold text-sm"
                             style={{ color: selected.color }}>
                          {selected.lat.toFixed(4)}°N
                        </div>
                      </div>
                      <div className="flex-1 p-2 rounded-lg text-center"
                           style={{
                             background: "rgba(255,255,255,0.04)",
                             border:     "1px solid rgba(255,255,255,0.07)",
                           }}>
                        <div className="text-[9px] font-mono text-white/30">LONGITUDE</div>
                        <div className="font-mono font-bold text-sm"
                             style={{ color: selected.color }}>
                          {selected.lon.toFixed(4)}°E
                        </div>
                      </div>
                    </div>

                    {/* Description */}
                    <p className="text-xs leading-relaxed mb-4"
                       style={{ color: "rgba(226,232,240,0.5)" }}>
                      {selected.desc}
                    </p>

                    {/* Navigation buttons */}
                    <div className="grid grid-cols-2 gap-2">
                      <button
                        onClick={() => openNavigation(selected)}
                        className="flex items-center justify-center gap-1.5
                                   py-2.5 rounded-xl font-bold text-xs
                                   transition-all hover:scale-105"
                        style={{
                          background: `linear-gradient(135deg, ${selected.color}, #7c3aed)`,
                          color:      "white",
                          boxShadow:  `0 0 15px ${selected.color}30`,
                        }}
                      >
                        <Navigation size={14} />
                        Navigate There
                      </button>
                      <button
                        onClick={() => openInMaps(selected)}
                        className="flex items-center justify-center gap-1.5
                                   py-2.5 rounded-xl font-bold text-xs
                                   transition-all hover:scale-105"
                        style={{
                          background:  `${selected.color}12`,
                          border:      `1px solid ${selected.color}30`,
                          color:        selected.color,
                        }}
                      >
                        <Globe size={14} />
                        Open in Maps
                      </button>
                    </div>

                    {/* Save offline */}
                    <button
                      onClick={() => saveOffline(selected)}
                      className="w-full mt-2 flex items-center justify-center
                                 gap-1.5 py-2 rounded-xl text-xs font-mono
                                 transition-all hover:scale-105"
                      style={{
                        background: "rgba(255,255,255,0.03)",
                        border:     "1px solid rgba(255,255,255,0.08)",
                        color:      "rgba(226,232,240,0.4)",
                      }}
                    >
                      <Download size={12} />
                      Save for Offline Use
                    </button>
                  </div>

                  {/* Navigation metrics */}
                  {bearing !== null && distance !== null && (
                    <div className="rounded-2xl p-5"
                         style={{
                           background:     "rgba(0,0,8,0.85)",
                           backdropFilter: "blur(20px)",
                           border:         `1px solid ${selected.color}20`,
                         }}>
                      <div className="flex items-center justify-between mb-4">
                        <div className="font-bold text-white text-sm flex items-center gap-2">
                          <Compass size={16} style={{ color: selected.color }} />
                          Live Navigation
                        </div>
                        <div className="font-mono font-bold text-lg"
                             style={{ color: selected.color }}>
                          {distance.toLocaleString()} km
                        </div>
                      </div>

                      <div className="flex items-center justify-between">
                        <CompassDisplay
                          bearing={bearing}
                          color={selected.color}
                        />
                        <div className="text-right space-y-3 ml-6">
                          <div>
                            <div className="text-[10px] font-mono text-white/30">BEARING</div>
                            <div className="font-mono font-bold text-2xl"
                                 style={{ color: selected.color }}>
                              {bearing}°
                            </div>
                          </div>
                          <div>
                            <div className="text-[10px] font-mono text-white/30">DIRECTION</div>
                            <div className="font-mono font-bold text-xl"
                                 style={{ color: selected.color }}>
                              {bearingToCompass(bearing)}
                            </div>
                          </div>
                          <div>
                            <div className="text-[10px] font-mono text-white/30">DISTANCE</div>
                            <div className="font-mono font-bold text-base text-white">
                              {distance > 1000
                                ? `${(distance / 1000).toFixed(1)}K km`
                                : `${distance} km`}
                            </div>
                          </div>
                        </div>
                      </div>
                    </div>
                  )}

                  {/* Tab selector */}
                  <div className="flex gap-1 p-1 rounded-xl"
                       style={{
                         background: "rgba(0,0,8,0.8)",
                         border:     "1px solid rgba(255,255,255,0.06)",
                       }}>
                    {[
                      { id: "navigate" as const,  label: "Approach"   },
                      { id: "protocol" as const,  label: "Frequencies"},
                      { id: "protect"  as const,  label: "Protection" },
                    ].map(tab => (
                      <button
                        key={tab.id}
                        onClick={() => setActiveTab(tab.id)}
                        className="flex-1 py-2 rounded-lg text-xs font-medium
                                   transition-all"
                        style={{
                          background: activeTab === tab.id
                            ? `${selected.color}15` : "transparent",
                          color: activeTab === tab.id
                            ? selected.color : "rgba(226,232,240,0.35)",
                          border: activeTab === tab.id
                            ? `1px solid ${selected.color}30` : "1px solid transparent",
                        }}
                      >
                        {tab.label}
                      </button>
                    ))}
                  </div>

                  <AnimatePresence mode="wait">

                    {/* Approach tab */}
                    {activeTab === "navigate" && (
                      <motion.div
                        key="navigate"
                        initial={{ opacity: 0, y: 10 }}
                        animate={{ opacity: 1, y: 0 }}
                        exit={{ opacity: 0, y: -10 }}
                        className="rounded-2xl p-5 space-y-2"
                        style={{
                          background:     "rgba(0,0,8,0.85)",
                          backdropFilter: "blur(20px)",
                          border:         `1px solid ${selected.color}18`,
                        }}
                      >
                        <div className="font-bold text-white text-sm mb-3 flex items-center gap-2">
                          <MapPin size={14} style={{ color: selected.color }} />
                          Approach Protocol
                        </div>
                        {selected.approach.map((step, i) => (
                          <label
                            key={i}
                            className="flex items-start gap-3 cursor-pointer
                                       p-2 rounded-xl transition-all hover:bg-white/3"
                            onClick={() => toggleCheck(`approach_${i}`)}
                          >
                            <div
                              className="w-5 h-5 rounded flex items-center
                                         justify-center flex-shrink-0 mt-0.5"
                              style={{
                                background: checklist[`approach_${i}`]
                                  ? selected.color : "rgba(255,255,255,0.06)",
                                border: `1px solid ${selected.color}40`,
                              }}
                            >
                              {checklist[`approach_${i}`] && (
                                <CheckCircle size={12} className="text-white" />
                              )}
                            </div>
                            <span className="text-xs leading-relaxed"
                                  style={{
                                    color: checklist[`approach_${i}`]
                                      ? "rgba(226,232,240,0.4)"
                                      : "rgba(226,232,240,0.65)",
                                    textDecoration: checklist[`approach_${i}`]
                                      ? "line-through" : "none",
                                  }}>
                              {step}
                            </span>
                          </label>
                        ))}

                        {/* Warnings */}
                        <div className="mt-4 pt-3"
                             style={{ borderTop: `1px solid ${selected.color}12` }}>
                          <div className="text-[10px] font-mono font-bold mb-2 flex items-center gap-1"
                               style={{ color: "#ef4444" }}>
                            <AlertTriangle size={11} />
                            WARNINGS
                          </div>
                          {selected.warnings.map((w, i) => (
                            <div key={i} className="flex items-start gap-2 mb-1.5">
                              <AlertTriangle size={10}
                                            className="flex-shrink-0 mt-0.5"
                                            style={{ color: "#ef4444" }} />
                              <span className="text-[10px] leading-relaxed"
                                    style={{ color: "rgba(239,68,68,0.7)" }}>
                                {w}
                              </span>
                            </div>
                          ))}
                        </div>
                      </motion.div>
                    )}

                    {/* Frequencies tab */}
                    {activeTab === "protocol" && (
                      <motion.div
                        key="protocol"
                        initial={{ opacity: 0, y: 10 }}
                        animate={{ opacity: 1, y: 0 }}
                        exit={{ opacity: 0, y: -10 }}
                      >
                        <LocationTonePlayer location={selected} />

                        <div className="mt-3 p-3 rounded-xl"
                             style={{
                               background:  `${selected.color}08`,
                               border:      `1px solid ${selected.color}18`,
                             }}>
                          <div className="text-[10px] font-mono font-bold mb-1"
                               style={{ color: selected.color }}>
                            BRAIN LOCK FREQUENCY
                          </div>
                          <div className="text-2xl font-bold font-mono"
                               style={{ color: selected.color }}>
                            {selected.brainLockHz} Hz
                          </div>
                          <div className="text-[10px]"
                               style={{ color: "rgba(226,232,240,0.35)" }}>
                            {selected.brainLockHz === 8.22
                              ? "Antarctic Schumann — pole specific. NOT 7.83."
                              : "Standard Schumann — Earth heartbeat lock"}
                          </div>
                          <div className="text-[9px] font-mono mt-1"
                               style={{ color: "rgba(226,232,240,0.25)" }}>
                            ● = Brain lock frequency (play first and continuously)
                          </div>
                        </div>
                      </motion.div>
                    )}

                    {/* Protection tab */}
                    {activeTab === "protect" && (
                      <motion.div
                        key="protect"
                        initial={{ opacity: 0, y: 10 }}
                        animate={{ opacity: 1, y: 0 }}
                        exit={{ opacity: 0, y: -10 }}
                        className="rounded-2xl p-5 space-y-2"
                        style={{
                          background:     "rgba(0,0,8,0.85)",
                          backdropFilter: "blur(20px)",
                          border:         "1px solid rgba(168,85,247,0.2)",
                        }}
                      >
                        <div className="font-bold text-white text-sm mb-3 flex items-center gap-2">
                          <Shield size={14} style={{ color: "#a855f7" }} />
                          Protection Checklist
                        </div>
                        {selected.protection.map((item, i) => (
                          <label
                            key={i}
                            className="flex items-start gap-3 cursor-pointer
                                       p-2 rounded-xl transition-all hover:bg-white/3"
                            onClick={() => toggleCheck(`protect_${i}`)}
                          >
                            <div
                              className="w-5 h-5 rounded flex items-center
                                         justify-center flex-shrink-0 mt-0.5"
                              style={{
                                background: checklist[`protect_${i}`]
                                  ? "#a855f7" : "rgba(255,255,255,0.06)",
                                border: "1px solid rgba(168,85,247,0.4)",
                              }}
                            >
                              {checklist[`protect_${i}`] && (
                                <CheckCircle size={12} className="text-white" />
                              )}
                            </div>
                            <span className="text-xs leading-relaxed"
                                  style={{
                                    color: checklist[`protect_${i}`]
                                      ? "rgba(226,232,240,0.3)"
                                      : "rgba(226,232,240,0.65)",
                                    textDecoration: checklist[`protect_${i}`]
                                      ? "line-through" : "none",
                                  }}>
                              {item}
                            </span>
                          </label>
                        ))}

                        {/* Protection score */}
                        <div className="mt-4 pt-3"
                             style={{ borderTop: "1px solid rgba(168,85,247,0.12)" }}>
                          {(() => {
                            const total    = selected.protection.length;
                            const complete = Object.keys(checklist)
                              .filter(k => k.startsWith("protect_") && checklist[k])
                              .length;
                            const pct = Math.round((complete / total) * 100);
                            return (
                              <>
                                <div className="flex justify-between text-xs font-mono mb-1.5">
                                  <span style={{ color: "rgba(226,232,240,0.3)" }}>
                                    Protection Ready
                                  </span>
                                  <span style={{ color: pct === 100 ? "#22c55e" : "#a855f7" }}>
                                    {pct}%
                                  </span>
                                </div>
                                <div className="h-1.5 rounded-full overflow-hidden"
                                     style={{ background: "rgba(255,255,255,0.06)" }}>
                                  <motion.div
                                    className="h-full rounded-full"
                                    style={{
                                      background: pct === 100 ? "#22c55e" : "#a855f7",
                                    }}
                                    animate={{ width: `${pct}%` }}
                                    transition={{ duration: 0.5 }}
                                  />
                                </div>
                                {pct === 100 && (
                                  <div className="mt-2 text-center text-xs font-mono font-bold"
                                       style={{ color: "#22c55e" }}>
                                    ✓ PROTECTED — YOU MAY APPROACH
                                  </div>
                                )}
                              </>
                            );
                          })()}
                        </div>
                      </motion.div>
                    )}
                  </AnimatePresence>
                </>
              ) : (
                /* No selection state */
                <div className="rounded-2xl p-8 text-center"
                     style={{
                       background:     "rgba(0,0,8,0.8)",
                       backdropFilter: "blur(20px)",
                       border:         "1px solid rgba(251,191,36,0.15)",
                     }}>
                  <div className="text-4xl mb-4">🗺️</div>
                  <div className="font-bold text-white text-xl mb-2">
                    Select a Location
                  </div>
                  <p style={{ color: "rgba(226,232,240,0.35)" }} className="text-sm mb-6">
                    Click any pin on the globe
                    or select from the grid below.
                    GPS navigation, frequencies,
                    and protection protocols
                    load automatically.
                  </p>

                  <div className="space-y-2">
                    {[
                      { label: "9 Gate Locations Mapped",     icon: "📍" },
                      { label: "Live GPS Navigation",          icon: "🧭" },
                      { label: "Auto Frequency Load",          icon: "🎵" },
                      { label: "Protection Checklist",         icon: "🛡️" },
                      { label: "Works Offline",                icon: "📶" },
                      { label: "Open in Google Maps",          icon: "🗺️" },
                    ].map((item, i) => (
                      <div key={i}
                           className="flex items-center gap-3 px-3 py-2 rounded-xl"
                           style={{ background: "rgba(255,255,255,0.03)" }}>
                        <span>{item.icon}</span>
                        <span className="text-xs"
                              style={{ color: "rgba(226,232,240,0.5)" }}>
                          {item.label}
                        </span>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          </div>
        </div>
      </div>
    </main>
  );
}
